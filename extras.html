Shared components

Reusable pieces built on style.css. Each one below is the real, live component — not a screenshot — so you're always looking at current styles. Copy the markup under each example into any page.

Component

Stat Band

A full-bleed, high-contrast section for headline numbers — program results, counts, dollar amounts. No imagery; the numbers are the visual. Each number counts up from 0 when it scrolls into view (about 3 seconds, eased), and respects "reduce motion" settings by jumping straight to the final value. Works with any number of stats — the grid reflows from 2 columns on mobile up to 5 on desktop.

(function () { const reduceMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches; const DURATION = 3000; const easeOutCubic = (t) => 1 - Math.pow(1 - t, 3); function formatValue(n, el) { const decimals = parseInt(el.dataset.decimals || "0", 10); const prefix = el.dataset.prefix || ""; const suffix = el.dataset.suffix || ""; const body = decimals > 0 ? n.toFixed(decimals) : Math.round(n).toLocaleString("en-US"); return prefix + body + suffix; } function animateStat(el) { const target = parseFloat(el.dataset.count); if (reduceMotion || isNaN(target)) { el.textContent = formatValue(target || 0, el); return; } const start = performance.now(); function tick(now) { const p = Math.min((now - start) / DURATION, 1); el.textContent = formatValue(target * easeOutCubic(p), el); if (p < 1) requestAnimationFrame(tick); } requestAnimationFrame(tick); } // Reserve each number's final width (see .stat b::after in style.css). document.querySelectorAll("[data-count]").forEach((el) => { el.dataset.final = formatValue(parseFloat(el.dataset.count) || 0, el); }); const grids = document.querySelectorAll(".stat-grid"); if (!grids.length) return; const io = new IntersectionObserver((entries) => { entries.forEach((entry) => { if (!entry.isIntersecting) return; entry.target.querySelectorAll("[data-count]").forEach(animateStat); io.unobserve(entry.target); }); }, { threshold: 0.4 }); grids.forEach((g) => io.observe(g)); })(); </script>
Component

Tabs

An accessible tablist for switching between two or more audiences or views on one page (for example, “For Employers” and “For Youth”). Panels are hardcoded in the HTML; the script only toggles which one is visible. Supports arrow keys, Home/End, and deep links such as page.html#tab-youth. Put it on a .tint section or a white one; .tp-lead takes an optional .ph photo (drop the photo and add .solo).

First panel

Panel content goes here. Use .tp-lead for a lead block and .tp-cols for a two-column row beneath it.

Second panel

Only one panel is visible at a time. Arrow keys, Home, and End move between tabs.

Component

Photo Gallery

A responsive grid of photo thumbnails that opens a full-screen viewer (a native <dialog>) with previous/next buttons, arrow keys, Escape, and click-outside to close. Thumbnails are real <img> tags inside .ph, so hover zoom scales the image. Add .two for 2 columns and .wide for 2:1 thumbnails. Each page needs the one <dialog class="lightbox"> shown below, once, before the footer.
